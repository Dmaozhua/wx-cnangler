let audioContext = null

Page({
  // ...
  playMusic(e) {
    const index = e.currentTarget.dataset.index
    const music = this.data.musicList[index]
    
    if (!audioContext) {
      audioContext = wx.createInnerAudioContext()
      audioContext.onPlay(() => {
        wx.showToast({ title: '开始播放' })
      })
    }
    
    audioContext.stop()
    audioContext.src = music.url
    audioContext.play()
  }
})

2. 逻辑处理增强
javascript
// pages/music/player.js
Page({
  data: {
    currentTime: 0,
    duration: 0,
    playing: false
  },

  onLoad(options) {
    this.audioContext = wx.createInnerAudioContext()
    this.initAudioEvents()
  },

  initAudioEvents() {
    // 播放时间更新
    this.audioContext.onTimeUpdate(() => {
      this.setData({
        currentTime: this.audioContext.currentTime,
        duration: this.audioContext.duration || 0
      })
    })

    // 播放状态监听
    this.audioContext.onPlay(() => {
      this.setData({ playing: true })
    })
    
    this.audioContext.onPause(() => {
      this.setData({ playing: false })
    })
  },

  // 格式化时间显示
  formatTime(time) {
    const minutes = Math.floor(time / 60)
    const seconds = Math.floor(time % 60)
    return `${minutes}:${seconds.toString().padStart(2, '0')}`
  },

  // 拖动进度条处理
  onSliderChange(e) {
    const value = e.detail.value
    this.audioContext.seek(value)
    if (!this.data.playing) {
      this.audioContext.play()
    }
  },

  // 实时拖动中处理
  onSliderChanging(e) {
    this.setData({ currentTime: e.detail.value })
  },

  // 播放/暂停切换
  togglePlay() {
    if (this.data.playing) {
      this.audioContext.pause()
    } else {
      this.audioContext.play()
    }
  }
})

2. 歌词解析工具

// utils/lyric-parser.js
function parseLyric(text) {
  const lines = text.split('\n')
  const pattern = /\[(\d{2}):(\d{2})(\.\d{2,3})?\]/g
  const result = []

  lines.forEach(line => {
    const timeArr = []
    let lyricText = ''
    
    // 提取时间标签
    const matches = line.matchAll(pattern)
    for (const match of matches) {
      const minutes = parseFloat(match[1])
      const seconds = parseFloat(match[2])
      const milliseconds = match[3] ? parseFloat(match[3]) : 0
      const totalTime = minutes * 60 + seconds + milliseconds
      timeArr.push(totalTime)
      lyricText = line.replace(pattern, '')
    }

    // 创建歌词项
    if (timeArr.length > 0 && lyricText.trim()) {
      timeArr.forEach(time => {
        result.push({
          time: Number(time.toFixed(3)),
          text: lyricText.trim()
        })
      })
    }
  })

  // 按时间排序
  return result.sort((a, b) => a.time - b.time)
}

module.exports = parseLyric

// components/lyric-display.js
const parseLyric = require('../../utils/lyric-parser')

Component({
  properties: {
    currentTime: Number,
    lyricUrl: String
  },

  data: {
    lyrics: [],
    activeIndex: -1,
    scrollTop: 0
  },

  observers: {
    'lyricUrl'(url) {
      if (url) this.loadLyric(url)
    },
    'currentTime'(time) {
      this.updateLyric(time)
    }
  },

  methods: {
    async loadLyric(url) {
      const res = await wx.request({ url })
      const lyrics = parseLyric(res.data)
      this.setData({ lyrics })
    },

    updateLyric(currentTime) {
      const { lyrics } = this.data
      let activeIndex = -1
      
      // 查找当前对应的歌词行
      for (let i = 0; i < lyrics.length; i++) {
        if (currentTime >= lyrics[i].time) {
          activeIndex = i
        } else {
          break
        }
      }

      if (activeIndex !== this.data.activeIndex) {
        this.setData({ 
          activeIndex,
          scrollTop: activeIndex * 60 // 根据行高计算滚动位置
        })
      }
    }
  }
})

function findCurrentLine(time, lyrics) {
  let left = 0
  let right = lyrics.length - 1
  
  while (left <= right) {
    const mid = Math.floor((left + right) / 2)
    if (lyrics[mid].time <= time) {
      left = mid + 1
    } else {
      right = mid - 1
    }
  }
  return right
}
